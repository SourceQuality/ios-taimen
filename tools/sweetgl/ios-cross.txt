# meson cross file for iOS 16 arm64. The constants toolchain, sdk, sweetgl and
# cc_wrapper (['ccache'] when installed) come from out/ios-paths.txt, which
# mesa-setup.sh writes from xcrun and passes as the first --cross-file.
[binaries]
c = cc_wrapper + [toolchain / 'clang', '-arch', 'arm64', '-isysroot', sdk, '-miphoneos-version-min=16.0']
cpp = cc_wrapper + [toolchain / 'clang++', '-arch', 'arm64', '-isysroot', sdk, '-miphoneos-version-min=16.0']
objc = cc_wrapper + [toolchain / 'clang', '-arch', 'arm64', '-isysroot', sdk, '-miphoneos-version-min=16.0']
ar = toolchain / 'ar'
strip = toolchain / 'strip'
pkg-config = 'false'

[built-in options]
c_args = ['-I' + sweetgl / 'include']
cpp_args = ['-I' + sweetgl / 'include']

[host_machine]
system = 'darwin'
subsystem = 'ios'
cpu_family = 'aarch64'
cpu = 'arm64'
endian = 'little'
